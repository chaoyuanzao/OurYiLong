<script>
const tooltipHover = document.getElementById('tooltipHover');
const tooltipSelected = document.getElementById('tooltipSelected');
const selectedNameDom = tooltipSelected.querySelector('.name'); // 选中提示：区块名称DOM
const selectedDescDom = tooltipSelected.querySelector('.desc'); // 选中提示：简介DOM
const pieces = document.querySelectorAll('.piece');
let activeItem = null; // 保存当前选中区块

// 判断设备是否为触摸设备（手机/平板），触摸设备直接隐藏鼠标悬浮提示
const isTouchDevice = 'ontouchstart' in window || navigator.maxTouchPoints > 0;
if(isTouchDevice){
  tooltipHover.style.display = "none";
}

pieces.forEach(el => {
  // 鼠标悬浮临时提示（仅PC端生效）
  el.addEventListener('mouseenter', (e) => {
    tooltipHover.textContent = el.dataset.name;
    tooltipHover.style.display = 'block';
  });
  el.addEventListener('mousemove', (e) => {
    tooltipHover.style.left = (e.clientX + 12) + 'px';
    tooltipHover.style.top = (e.clientY + 12) + 'px';
  });
  el.addEventListener('mouseleave', () => {
    tooltipHover.style.display = 'none';
  });

  // 点击事件核心逻辑
  el.addEventListener('click', () => {
    if(activeItem === el){
      // 再次点击同一个：取消选中
      el.classList.remove('active');
      activeItem = null;
      tooltipSelected.style.display = 'none';
    }else{
      // 点击别的区块：清除旧选中，选中当前
      if(activeItem){
        activeItem.classList.remove('active');
      }
      el.classList.add('active');
      activeItem = el;
      // 读取名称和简介，渲染两行文字
      selectedNameDom.textContent = el.dataset.name;
      selectedDescDom.textContent = el.dataset.desc;
      tooltipSelected.style.display = 'block';
    }
  });
});
</script>
